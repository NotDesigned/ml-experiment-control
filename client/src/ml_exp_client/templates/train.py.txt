"""Dependency-free output/metrics smoke; use your own training code for research."""
import argparse
import json
import os
from pathlib import Path
import time

from ml_exp_metrics import MetricWriter

parser = argparse.ArgumentParser()
parser.add_argument("--steps", type=int, default=3)
parser.add_argument("--step-delay", type=float, default=0, help="seconds between smoke steps")
args = parser.parse_args()
output = Path(os.environ["OUTPUT_DIR"])
output.mkdir(parents=True, exist_ok=True)
metrics = MetricWriter(output)
for step in range(1, args.steps + 1):
    metrics.log("loss", 1 / step, unit="dimensionless", step=step)
    time.sleep(max(0, args.step_delay))
summary = {key.lower(): os.environ[key] for key in
           ("PROJECT_NAME", "RUN_ID", "ATTEMPT_ID", "SOURCE_ID")}
summary["steps"] = args.steps
(output / "summary.json").write_text(json.dumps(summary, indent=2) + "\n")
print(json.dumps(summary), flush=True)
